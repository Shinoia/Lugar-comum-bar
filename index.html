function sincronizar() { var urlFetch = API_URL; // Ignora o cache temporariamente para garantir leitura direta da planilha fetch(urlFetch) .then(function(res) { return res.json(); }) .then(function(data) { if (data) { if (data.produtos && data.produtos.length > 0) { CARDAPIO = data.produtos; localStorage.setItem('cardapio_lugar_comum', JSON.stringify(data.produtos)); localStorage.setItem('cardapio_tempo', new Date().getTime()); } if (data.pedidos) { estado.pedidosGlobais = data.pedidos; processarPedidos(); } renderizar(); } }) .catch(function(e) { console.log("Erro na rede:", e); var cached = localStorage.getItem('cardapio_lugar_comum'); if (cached) CARDAPIO = JSON.parse(cached); renderizar(); }); }
